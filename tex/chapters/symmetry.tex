This section develops the theory underlying \emph{symmetric tensor networks},
specifically for non-abelian groups. We give a quick overview over the necessary
representation theoretic concepts in \Cref{sym:sub:rep}. We employ the example
of the $XXZ$-Hamiltonian for both abelian and non-abelian symmetries and give an
exact calculation of its conserved quantities for lattices of size $2$ and $3$.
Here we are mostly interested in the \emph{speedup} obtained by exploiting the
emergent block-structure of symmetric operators. Most of the proofs and formal
definitions required for this formalism are delegated to \Cref{sect:Categories},
however there will still be repetitions of some of the theory.

\section{Representation theory}\label{sym:sub:rep}

This section gives a quick review of representation theory. We consider all
vector spaces and their representations here to be finite, the groups may only
be compact. For some of the more basic definitions as those of a group and some
of the matrix groups we are using here, see \Cref{app:gen:groups}. We follow the
books by \textcite{ibortIntroductionGroupsGroupoids2019},
\textcite{serreLinearRepresentationsFinite1977},
\textcite{woitQuantumTheoryGroups2017} \todo{cite woit book} as well as
\textcite{etingofIntroductionRepresentationTheory2011}.

\begin{definition}
  A \green{\emph{representation}} of a group $(G,\cdot)$ which is just $G$ here
  over a vector space $V$ is a group homomorphism
  \[
    \rho : G \to \End (V), \quad \rho(g)\rho(h) = \rho(g \cdot h), \quad \rho(e) = id_V
  \]
  The \green{\emph{representation space}} is denoted by the tuple $(\rho, V)$
  but usually we shorten this to just $V$.

  If we want the representation to be linear, we can choose it as
  \[
    \rho : G \to \GL(V)
  \]
  instead. Choosing a linear basis for $V$ in $\C^n$ gives this as 
  \[
    \rho : G \to \GL_n(\C).
  \]
\end{definition}


\begin{definition}\label{sym:def:intertwiner}
  A linear map $A : V \to W$ is an \emph{\green{intertwiner}}, if under
  representations $\rho_V : G \to \GL(V)$ and $\rho_W : G \to \GL(W)$ it holds
  that
  \[
    A \circ \rho_V(g) = \rho_W (g) \circ A \quad \forall g \in G.
  \]
  Denote the space of all intertwiners between $V$ and $W$ of the group $G$ as
  $\Hom_G(V,W)$, similarly denote its endomorphisms as $\End_G(V) \coloneqq
  \Hom_G(V,V)$.
\end{definition}

\begin{definition} If our vector space $V$ is equipped  with a (Hermitian?)
  inner product, or in other words we are given a representation $\pi : G \to
  \U{\mathcal{H}} \hookrightarrow \GL(\mathcal{H})$ where $\mathcal{H}$ is a
  complex Hilbert space, then $\pi$ is unitary if
  \[
    \pi(g)^\dagger \pi(g) = \pi(g)\pi(g)^\dagger = I.
  \]
  This directly entails that $\pi(g^{-1}) = \pi(g)^\dagger$.
\end{definition}

\begin{definition}
  \label{symm:def:tensor_product_representation}
  Given reps $(\rho_V, V)$ and $(\rho_W, W)$, the \emph{\green{tensor product
  representation}} $(\rho_{V\otimes W}, V \otimes W)$ is defined by its action
  on the elements
  \[
    \rho_{V \otimes W}(g) (v \otimes w) = \rho_V(g) v \otimes \rho_W(g) w
  \]
\end{definition}

\begin{definition}
  As above, the \emph{\green{direct sum representation}} $(\rho_{V \oplus W}, V
  \oplus W)$ 
  \[
    \rho_{V \oplus W}(g) = \begin{pmatrix}
      \rho_V(g) &   \\
       &  \rho_W(g) \\
    \end{pmatrix}
  \]
\end{definition}

\begin{definition}[Dual representation, cf. {\cite[~§
1.1]{fultonRepresentationTheory2004}}] Each representation $(\rho, V)$ dual
space $V^\ast$ (\eqref{???dualspace???}) also acts as a representation
$\rho^\ast : G \to \GL(V^\ast)$ and so as a linear functional
\begin{equation}\label{sym:eq:dualRepresentation}
  \rho^\ast(g) : f \mapsto f \circ \rho(g^{-1}).
\end{equation}
\end{definition}


\begin{definition}
  A representation $\rho$ is called \emph{\green{irreducible}} if it cannot be written as a direct sum representation.

  If $G$ is finite, we can group these irreducibles (irreps) into isomorphism classes, and we denote these
  \[
    \Irr(G) \cong \{[\rho_i]\}_{i \in I}. 
  \]
\end{definition}

\begin{theorem}
  There is a relation on the dimensions of the representations (cf. \cite[~Thm
  3.1 (ii)]{etingofIntroductionRepresentationTheory2011})
  \[
    \sum_{i \in \Irr(G)} (\dim(\rho_i))^2 = |G|.
  \]
\end{theorem}

\begin{lemma}[Schur's Lemma, cf. {
  \cite{deutscheakademiederwissenschaftenzuberlinSitzungsberichteKoeniglichPreussischen1882}
  \cite[~Thm. 7.5]{ibortIntroductionGroupsGroupoids2019}}]
  \label{sym:lemma:schur}
  \emph{\green{Schur's} Lemma} states that for irreps $A$ and $B$ of $G$, we can
  give its endomorphisms as a complex multiple of the identity (cf. ). Further
  it holds that
  \begin{equation}\label{sym:eq:schur} 
    \Hom_G(A,B) = \begin{dcases}
      \C \cdot \id_A, &\text{ if } A \cong B ;\\
      0 , &\text{ if } A \ncong B .
    \end{dcases}
  \end{equation}
  By isomorphism and basis choice, we can directly express this as a matrix
  representation
  \[
    \C \cdot \id_A \cong \C \ot \id_A \cong \id_A \ot\; \C \cong \id_{\dim(A)} \ot\; \C.
  \]
\end{lemma}

\begin{definition}[Canonical decomposition {(cf. \cite[~§2.6, Thm.
  8]{serreLinearRepresentationsFinite1977} \cite[~Prop.
  1.8]{fultonRepresentationTheory2004})}]\label{sym:def:isotypic} A
  representation $V$ is semisimple (or isotypic) if it decomposes as
  \[
    V \cong \bigoplus_{i \in I} V_i^{\oplus n_i} 
    \cong 
    \bigoplus_{i \in I} \C^{n_i} \otimes V_i
    \cong
    \bigoplus_{i \in I} V_i \otimes \C^{n_i}
  \]
  by grouping isomorphic simples. Here we call $n_i$ the multiplicity of its
  corresponding irrep and $\C^{n_i}$ the \emph{multiplicity-space}. Canonically,
  we can define the multiplicity space as the vector space $\Hom_G(V_i, V)$
  \cite[~Cor. 2.16]{fultonRepresentationTheory2004}.
\end{definition}

These maps are more commonly known as \emph{Clebsch-Gordan tensors}. Note that
the equations 
\begin{equation}
  \bra{\nu, d; a,b} \ket{a,b ; c,\mu} = \delta_{\mu \nu} \delta_{c d} \id_{R_c} 
\end{equation}
and 
\begin{equation}
  \sum_{c,\mu}^{} \ket{a,b;c,\mu}\bra{\mu,c; a,b} = \id_{R_a \otimes R_b}.
\end{equation}
hold. As a consequence of this relation, the outer product of two Clebsch-Gordan
spans the space $\End_G(R_a \otimes R_b)$.

\section{Application: The XXZ Hamiltonian}

The motivation for constructing symmetric tensor networks is as follows: Assume
we have a (unitary) representation of a group $G$ denoted $U(g)$.  Furthermore
assume we have a Hamiltonian that commutes with $[U(g)^{\otimes L}, H] = 0$,
equivalently, $H$ admits a \emph{global symmetry} $U(g)^{\otimes L} H
(U(g)^{\otimes L})^\dagger$. Then there are different presentations of $H$ with
respect to a certain basis by \emph{simultaneous
diagonalization}\cite[Thm~1.3.12]{Horn_Johnson_2012}, as shown in
\Cref{fig:diagonal_hamiltonians}. Notably these presentation induce a
\emph{block-sparse} structure on the respective operators

If we consider the XXZ model \eqref{def:xxzHamiltonian} with $\Delta=1$ (so the
XXX model), a nonabelian $\SU{2}$ symmetry emerges. Here, the generators of
$\SU{2}$ are given by the total spin operators for $x,y,z$ (cf. \cite[~§

\[
  S_{tot}^\alpha 
  =
  \sum_{i = 1}^{N} S_i^\alpha, \quad \alpha \in \{x,y,z\},
\]
and its representation is  $U(g_{\vec{n}}) = e^{i\theta \vec{n} \cdot S_{tot}}$.
The Hamiltonian is invariant under
\[
  U(g) H U(g)^\dagger = H, \quad g \in \SU 2
\]
which is equivalent to $[H, S_{tot}^\alpha] = 0$ for all $\alpha$. Since
\[
  [S_{tot}^x, S_{tot}^y ] = i S_{tot}^z 
\] (and all its permutations)
we cannot diagonalize these operators simultaneously and we need to consider a
new operator. The \emph{Casimir operator}
\[ 
  S_{tot}^2 = \sum_{a \in \{x,y,z\}}^{} (S_{tot}^{a})^2
\]
commutes with all the generators by the Lie-product rule

Or we can again do the representation theoretical approach. Let $V_j$ denote the
spin-$j$ irrep of $\SU 2$. We have that $\C^2 \cong V_{\frac{1}{2}}$ and by
\cref{sym:eq:cgseries} that
\[
  (\C^2)^{\ot 6} 
  \cong 
  V_{\frac{1}{2}}^{\ot 6} 
  \cong 
  5 V_0 \op 9V_1 \op 5V_2 \op V_3 
  \cong
  (\C^5 \ot V_{0}) \op (\C^9 \ot V_1) \op (\C^5 \ot V_2) \op (\C \ot V_3)
\]

Now our operator is a morphism in $\End_G(V_{\frac{1}{2}}^{\otimes 6})$ and thus
by Schur's lemma \eqref{sym:eq:schur}
\begin{equation}\label{sym:eq:blockdiag}
  H 
  \cong
  \bigoplus_{S \in \{0,1,2,3\}} 
  \Mat{m_S}{m_S}\C 
  \otimes I_{2S+1}
  \cong 
  \bigoplus_{S \in \{0,1,2,3\}} 
  I_{2S+1} \otimes \Mat{m_S}{m_S}\C .
\end{equation}
So while we only have to store the matrix elements as blocks, i.e. here we have
to store only four of them, in our picture we get 16 blocks because of the
multiplicities. If we were to store this Hamiltonian as its initial matrix, we
would have to store $64 \times 64 = 4096$ floats. If we only store the $4$
blocks, this is reduced to $5^2 + 9^2 + 5^2 + 1^2 = 132$ floats\footnote{As we
will see when discussing the actual algorithms behind these symmetric tensors,

calculations in \cref{sect:Categories}.}. If we take the case that $N$ is even
larger, i.e. the case $N = 10$, the full operator requires $1\,048\,576$ floats
while the symmetric operator only needs $16\,796$. Generally we have about two
orders of magnitude less.\todo{Do exact asymptotics here!}

While this big picture overview conveys the general idea, we should do some of

coefficients in $\cref{sym:eq:blockdiag}$, we restrict ourselves to the case of
the XXZ-Hamiltonian for $N = 2$ spins:
\[
  H = J \cdot (S_1^x S_2^x + S_1^y S_2^y + S_1^z S_2^z) = J \cdot S_1 \cdot S_2
\]

This acts on the space $V_{\frac{1}{2}} \otimes V_{\frac{1}{2}}$ and thus on the
basis $\{\ket{\uparrow},\ket{\downarrow}\}^{\otimes 2}$ This has two distinct
fusion outcomes, $V_0$ and $V_1$. The basis of the coupled states is given by

basis. We can do this purely algebraically by rearranging
\[
  S^2 = S_1^2 + S_2^2 + 2(S_1 \cdot S_2) \implies S_1 \cdot S_2 = \frac{1}{2}(S^2 - S_1^2 - S_2^2).
\]
Since $S_i^2 = s(s+1)$ is fixed for $s = \frac{1}{2}$ here, we ought to write
\[
  H = \frac{J}{2}(S^2 - 3/2)
\]
and this again splits into the sectors for $S = 0$ and $S = 1$
\[
  c_0 = \frac{J}{2}(0 - 3/2) = -\frac{3}{4}J

  c_1 = \frac{J}{2}(2 - 3/2) = \frac{1}{4}J
\]

expect our operator, now in $\bop\limits_{S \in \{0,1\}} \End_G(V_S)$ to look

\[

\section{Symmetric Tensors}

So far we have applied the symmetric formalism to Hamiltonians, which are
endomorphisms $\End_G()$ between some finite vector spaces, and we have an
interpretation of local Hamiltonians acting between sites. This section will
show how to express arbitrary symmetric tensors as intertwiners, and thus also
constructing symmetric MPSs, MPOs and MERAs.

Let $G$ be a group, and all representations thereof are
finite-dimensional.\footnote{Compactness of $G$ is not needed here; but we
require it later for Schur's lemma. Or where?} Fix representations $\rho_i\colon
G\to\GL{V_i}$ for $i=1,\dots,n$ and $\sigma_j\colon G\to\GL{W_j}$ for
$j=1,\dots,m$, and set
\begin{align*}
  V &= V_1\otimes\cdots\otimes V_n, &
  \rho_V(g) &= \rho_1(g)\otimes\cdots\otimes\rho_n(g),\\
  W &= W_1\otimes\cdots\otimes W_m, &
  \rho_W(g) &= \sigma_1(g)\otimes\cdots\otimes\sigma_m(g).
\end{align*}
The dual representation on $W_j^\ast$ is $\sigma_j^\ast(g)f =
f\circ\sigma_j(g)^{-1}$. A tensor with $n$ outgoing and $m$ incoming legs is an
element
\[
  T\in V_1\otimes\cdots\otimes V_n\otimes W_m^\ast\otimes\cdots\otimes W_1^\ast
  \;\cong\; V\otimes W^\ast,
\]
on which $G$ acts by
\begin{equation}\label{sym:eq:Pi}
  \Pi(g)=\rho_1(g)\otimes\cdots\otimes\rho_n(g)\otimes
         \sigma_m^\ast(g)\otimes\cdots\otimes\sigma_1^\ast(g).
\end{equation}

Recall the definition of an intertwiner \cref{sym:def:intertwiner} as an element
of $\Hom_G(-,-)$. For standard tensors, we had the equivalence between an object
of $V \ot W^\ast$ and the maps in $\Hom(V,W)$. Here we wish to show the same for
symmetric tensors.


\begin{definition}[Symmetric tensor
  \cite{singhTensorNetworkDecompositions2010a}]\label{def:symmetricTensor} A
  tensor $T\in V\otimes W^\ast$ is \emph{\green{symmetric}} (or $G$-invariant)
  if it remains unchanged under a global application of a representation
  $\Pi(g)$ on all of it legs, so  $\Pi(g)T=T$ for all $g\in G$, 
\end{definition}

Let $\Phi\colon V\otimes W^\ast\to\Hom(W,V)$ be the linear isomorphism
$\Phi(v\otimes f)=\bigl(w\mapsto f(w)\,v\bigr)$ as we have defined in
\cref{sect:background}.\todo{Dirac notation, correct label.}

\begin{lemma}\label{sym:lem:invariant_iff_intertwiner} For all $g\in G$ and
  $T\in V\otimes W^\ast$,
  \[
    \Phi\bigl(\Pi(g)T\bigr)=\rho_V(g)\circ\Phi(T)\circ\rho_W(g)^{-1}.
  \]
  Then $T$ is symmetric iff. $\Phi(T)\in\Hom_G(W,V)$, i.e.
  \[
    \rho_V(g)\circ\Phi(T)=\Phi(T)\circ\rho_W(g)\qquad\forall g\in G.
  \]
\end{lemma}
\begin{proof}
  Both sides are linear in $T$, so take $T=v\otimes f$. Then
  $\Pi(g)T=\rho_V(g)v\otimes\bigl(f\circ\rho_W(g)^{-1}\bigr)$, and
  \[
    \Phi(\Pi(g)T)(w)=f\bigl(\rho_W(g)^{-1}w\bigr)\,\rho_V(g)v
    =\rho_V(g)\,\Phi(T)\bigl(\rho_W(g)^{-1}w\bigr),
  \]
  which is the first claim. Since $\Phi$ is injective, $\Pi(g)T=T$ holds iff
  $\rho_V(g)\Phi(T)\rho_W(g)^{-1}=\Phi(T)$, and multiplying on the right by the
  invertible $\rho_W(g)$ gives the second claim.
\end{proof}

Hence $V\otimes W^\ast \cong\Hom_G(W,V)$, and we identify $T$ with $\Phi(T)$.
