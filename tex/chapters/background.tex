\begin{literature}[\bf{Anyons}]\label{Lit:Anyons}
  \emph{Anyons} as quasiparticles with exotic exchange statistics were originally propsed by \cite{wilczek1990} \cite{leinaasTheoryIdenticalParticles1977a}. Anyons are usually described as \emph{defects} in a (topologically orderd?) medium of dimension two (technically codimension after \cite{SS23AnyonsInTedK}).
  The exchange statistics of $N$ particles on this medium $M$ is goverend by the \emph{configuration space}
  \[
    \mathrm{Conf}_N(M) = \{(x_1,\ldots,x_N) \in M^N : x_i \neq x_j \} \setminus S_N.
  \]
  The statistics of the particle exchange is given by the representation of $\pi_1(\mathrm{Conf}_N(M))$ (the fundamental (homology?) group). If $M = \R^2$, this corresponds to the \emph{braid group} on $N$ strands $B_N$. If the dimension of $M \geq 3$ this corresponds to the symmetric group $S_N$. If these representations are unitary and the (ground state?) is degenerate, this allows for \emph{topological quantum computation} (cf. \Cref{lit:tqc}).
\end{literature}

\section{On Tensors}

Throughout, $k$ denotes a field and all vector spaces are finite-dimensional
over $k$.


A \emph{tensor} is an element of a tensor product of finite-dimensional vector
spaces,
\[
  T \in \bigotimes_{j \in J} V_j ,
\]
and we call $T$ a rank $\left\vert J \right\vert$ tensor. The \emph{dimension}
of the tensor is then $\dim(T) = \prod_{j \in J} \dim(V_j)$.

The \emph{dual space} of a vector space $V$ is $V^* = \operatorname{Hom}(V,k)$,
the space of linear functionals $f \colon V \to k$, $x \mapsto f(x)$. In Dirac
notation we write $\ket{v} \in V$ for vectors (kets) and $\bra{f} \in V^*$ for
linear functionals (bras), so that $\braket{f}{v} = f(v)$.

Since all spaces are finite-dimensional, we have the canonical isomorphisms
\[
  V^{**} \cong V, \qquad
  (V \otimes W)^* \cong V^* \otimes W^*, \qquad
  W \otimes V^* \cong \operatorname{Hom}(V,W),
\]
where the last is given on simple tensors by
\[
  \ket{w} \otimes \bra{f} \;\longmapsto\; \bigl(v \mapsto \braket f  v\,\ket w\bigr),
\]
i.e.\ $\ket{w}\!\bra{f}$ is the outer product.

Because $V^{**} \cong V$, any tensor $T \in V_1 \otimes \cdots \otimes V_n$ can
be regarded as a linear map after choosing a split of its legs at some $0 \le m
\le n$:
\[
  T : V_1^\ast \ot \cdots \ot V_{m}^\ast \longrightarrow V_{m+1} \ot \cdots \ot V_{n}.
\]

\begin{example}
  
 Let $t \in V_1 \otimes V_2 \otimes W_2^* \otimes W_1^*$.
Choosing the split after $V_2$, $t$ is an operator $W_1 \otimes W_2 \to V_1
\otimes V_2$, which in Dirac notation reads
\[
  t = \sum_{i_1,i_2,j_1,j_2} T^{i_1 i_2}_{j_1 j_2}\,
      \ket{v^1_{i_1}}\ket{v^2_{i_2}}\bra{w^2_{j_2}}\bra{w^1_{j_1}},
\]
where $\{\ket{v^a_i}\}$ is a basis of $V_a$ and $\{\bra{w^b_j}\}$ is a basis of
$W_b^*$ (the dual basis of a basis of $W_b$).
\end{example}

\begin{example}
  Note that while we gave a definition of a vector as $\ket{v} \in V$, we can alternatively define a vector as a linear map
  \[
    v : \C \to V, \lambda \mapsto \lambda \ket{v}.
  \]
  This means we can effectively let the space of vectors be the space
  $\Hom(\C,V)$.
\end{example}

Two tensors with matching legs are contracted using the \emph{evaluation}
and \emph{coevaluation} maps
\begin{align*}
  \mathrm{ev}_V &\colon V^* \otimes V \to k, & f \otimes v &\mapsto f(v),\\
  \mathrm{coev}_V &\colon k \to V \otimes V^*, & \lambda &\mapsto
    \lambda \sum_i v_i \otimes f_i ,
\end{align*}
where $\{v_i\}$ is a basis of $V$ and $\{f_i\}$ its dual basis. A
\emph{contraction} is a composite of identities and evaluation maps that
removes the paired legs from the tensor product. For example, for operators
$A \in U \otimes V^*$ (a map $V \to U$) and $B \in V \otimes W^*$ (a map
$W \to V$), the product $AB \in U \otimes W^*$ is
\[
  AB = (\mathrm{id}_U \otimes \mathrm{ev}_V \otimes \mathrm{id}_{W^*})(A \otimes B).
\]
Equip $U,V,W$ with bases $\{u_a\}, \{v_i\}$ and ${w_b}$ respectively. We can now
write
\[
  A \ot B = \sum_{a,i,j,b} A_{ai} B_{jb} \ket{u_a} \ot \bra{v_i} \ot \ket{v_i} \ot \bra{w_b}
\]
and by evaluation $\ev_V(\bra{v_j} \ot \ket{v_i}) = \braket{v_j}{v_i} =
\delta^i_j$ we can obtain
\[
  AB = \sum_{a,i,b} A_{ai} B_{jb} \ket{u_a} \ot \bra{w_b}
\]
with coordinates $(AB)^a_b = \sum_i A^a_i B^i_b$ which is precisely the well
known matrix-multiplication formula.

\subsection{Tensor Networks \& Matrix Product States}

Often times we will consider quantum state on a lattice. For $1$-dimensional
systems, we can consider a lattice of $n$ \emph{sites} where each holds a
\emph{local site} $\ket{i}$. We can give an arbitrary state on this lattice as
\[
  \ket{\Psi} = \sum_{i_1 \ldots i_n} c_{i_1 \cdots i_n} \ket{i_1 i_2 \cdots i_n}.
\]
Storing this state requires representing the exponentially many coefficient
$c_{i_1 \cdots i_n}$. As an example, we may have two sites of spin $1/2$, so we
have a total state space of dimension $4$ spanned by $\ket{i_1 i_2} \cong \C^4$
with $i_1,i_2 \in \{\uparrow, \downarrow\}$. Then, we can give the coefficients
to obtain the state
\[
  \Psi^+ = \frac{1}{\sqrt{2}}\left(\ket{\uparrow\uparrow} + \ket{\downarrow\downarrow}\right)
\]
and we can store this state as a $4\times1$ matrix (a statevector of dimension
4). If we keep the spin basis on the lattice sites, a matrix on $n$ sites is
thus representable by a statevector $\ket{\Psi} \in \C^{2^n}$.

Now, we are allowed to apply a reshape operation and change $\C^{2^n}$ into
$\C^{2 \times 2^{n-1}}$ which corresponds to a tensor of order $2$, i.e. a
matrix with $2$ columns and $2^{n-1}$ rows. Now that we have a matrix, we can apply the \gls{svd} yielding a decomposition 
\[
  c_{i_1, (i_2 \ldots i_n)} = \sum_{j_1} U_{i1,j1} S_{j_1,j_1} (V^\dagger)_{j1, (i_2 \ldots i_n)}
\]
Recall that we can choose the rank of the SVD, so that the matrix $U$ that is
produced is of dimension $r \leq i_1$\todo{NOTATION!}. We can absorb the factors $S$ and $V$ into the remaining state and again reshape to a state, and we get the expression
\[
  c_{i_1,(i_2 \ldots i_n)} = \sum_{j_1} U_{i_1,j_1} c_{j_1, i_2 \ldots i_n}
\]
We now reshape the new c again to a tensor of shape $[(j_1i_2, i_3 \ldots i_n)]$.
\[
  c_{i_1, i_2, \ldots} = \sum_{j_1, j_2} U^{i_1,j_1} U_{(j_1 i_2),j_2} c_{j_2, (i_3, \ldots, i_n)}
\]
and we can reshape the expression $U_{(j_1i_1),j_2}$ to a trivalent tensor having an extra leg for the physical dimension, i.e. we want ot build the tensor
\[
  c_{i_1, i_2, \ldots} = \sum_{j_1, j_2} U_{j_1}^{i_1} U_{j_1,j_2}^{i_2} c_{j_2, (i_3, \ldots, i_n)}
\]
We can apply this procedure recursively to the new coefficient, and in the end
we will get an expression for our state
\[
  c_{i_1, \ldots, i_n} = \sum_{j_1, \ldots, j_{n-1}} U_{i_1, j_1}
\]



\begin{definition}\label{def:MPS}
  A \emph{\green{Matrix Product State}} is a quantum state $\ket{\psi}$ with $n$ legs written as
  the contraction of $n$ trivalent tensors $\{A^{[i]}\}_{i \in [n]}$. Formally this reads
  \[
    \ket{\psi[A]}
    =
    \sum_{i_1,\ldots,i_n}
    \operatorname{Tr}\!\left(
    A^{i_1}
    A^{i_2}
    \cdots
    A^{i_n}
    \right)
    \ket{i_1i_2\cdots i_n},
  \]
  where each $A$ is of the form
  \[
    A^{[i]} = \sum_{i \delta_i \delta_{i+1}} A^{i}_{\delta_i \delta_{i+1}} \ket{i} \ot \bra{\delta_i} \ot \bra{\delta_{i+1}}
  \]
  where $A^{[i]}_{i_{k}} \in \C^{D_i \times D_{i+1} }$. The elements of the
  collection $\{D_i\}_{i \in [n]}$ is referred to as the
  \emph{\green{bond-dimensions}}.
\end{definition}
Note that this is a state with closed boundaries. If we want open boundaries we
have to replace the trace with appropriate end vectors (nice source?).

\begin{example}
  We can represent each product state $\ket{\varPsi} = \ket{\psi}^{\otimes n}$
  as a MPS with bond dimension $1$.
\end{example}

\begin{definition}

    \sum_{i_1, .., i_n j_1, ..., j_n} \tr(M^{i_1 j_1} M^{i_2 j_2} \cdots M^{i_n j_n}) \ket{i_1 \cdots i_n} \bra{j_1 \cdots j_n}
  \]
  We again use the convention of denoting the bonds by $D$, and we can regard each mps tensor as a mapping
  \[
    \Hom(P_i \otimes D_i, P_j \otimes D_{i+1})
  \]
\end{definition}

Constructing these operators can be done by the use of a \emph{finite state machine} \cite{crosswhiteFiniteAutomataCaching2008}. As an example, the XXZ-Hamiltonian is represented by the MPO tensors 
\begin{equation}
  W_{ij} = \begin{pmatrix}
    1 & 0 & 0 & 0 & 0 \\
    S^x & 0 & 0 & 0 & 0 \\
    S^y & 0 & 0 & 0 & 0 \\
    S^z & 0 & 0 & 0 & 0 \\
    0 & S^x & S^y & S^z & 1
  \end{pmatrix}_{ij}
\end{equation}
with bond dimension $5$, note that this acts on the physical states as such.
